**1. Cursor pagination helper (`internal/api/cursor.go`):**

```go
package api

import (
    "encoding/base64"
    "encoding/json"
    "net/http"
    "strconv"
)

type Cursor struct {
    ID        int64  `json:"i"`
    SortValue string `json:"s"`
}

func EncodeCursor(c Cursor) string {
    b, _ := json.Marshal(c)
    return base64.URLEncoding.EncodeToString(b)
}

func DecodeCursor(s string) (Cursor, error) {
    b, err := base64.URLEncoding.DecodeString(s)
    if err != nil {
        return Cursor{}, err
    }
    var c Cursor
    return c, json.Unmarshal(b, &c)
}

type PaginationParams struct {
    Limit  int
    Cursor *Cursor
}

const (
    DefaultPageLimit = 50
    MaxPageLimit     = 200
)

func ParsePaginationParams(r *http.Request) PaginationParams {
    p := PaginationParams{Limit: DefaultPageLimit}
    if v := r.URL.Query().Get("limit"); v != "" {
        if n, err := strconv.Atoi(v); err == nil && n > 0 && n <= MaxPageLimit {
            p.Limit = n
        }
    }
    if v := r.URL.Query().Get("cursor"); v != "" {
        if c, err := DecodeCursor(v); err == nil {
            p.Cursor = &c
        }
    }
    return p
}
```

**2. FTS5 UNION search (`internal/metadata/search.go`):**

```go
package metadata

import (
    "context"
    "database/sql"
    "strings"
)

type SearchResult struct {
    Kind     string  // "repo", "artifact", "cve", "rpm", "deb", "pypi", "helm"
    EntityID int64
    Name     string
    Location string  // project/repo path
    Severity string  // for CVE results
    Score    float64
}

type SearchParams struct {
    Query    string
    Kind     string // filter: empty = all
    Severity string // filter: empty = all
    Project  string // filter: empty = all
    Limit    int
    Offset   int
}

func (db *DB) SearchAll(ctx context.Context, p SearchParams) ([]SearchResult, error) {
    // Sanitize FTS5 query: escape double-quotes, wrap terms
    ftsQuery := sanitizeFTS5Query(p.Query)
    if ftsQuery == "" {
        return nil, nil
    }

    // Build UNION ALL across all 7 FTS5 tables with per-arm LIMIT
    // to prevent any single table from dominating results
    arms := []string{}
    args := []any{}
    perArmLimit := p.Limit * 2 // over-fetch per arm, trim after sort

    addArm := func(kind, query string) {
        arms = append(arms, query)
        args = append(args, ftsQuery, perArmLimit)
    }

    if p.Kind == "" || p.Kind == "repo" {
        addArm("repo", `SELECT 'repo' AS kind, rowid AS entity_id, name, project_name AS location, '' AS severity, rank * 1.5 AS score FROM repos_fts WHERE repos_fts MATCH ? LIMIT ?`)
    }
    if p.Kind == "" || p.Kind == "artifact" {
        addArm("artifact", `SELECT 'artifact', rowid, artifact_name, version AS location, '' AS severity, rank FROM artifacts_fts WHERE artifacts_fts MATCH ? LIMIT ?`)
    }
    if p.Kind == "" || p.Kind == "cve" {
        addArm("cve", `SELECT 'cve', rowid, cve_id, package AS location, title AS severity, rank FROM cves_fts WHERE cves_fts MATCH ? LIMIT ?`)
    }
    // Package-specific FTS tables
    if p.Kind == "" || p.Kind == "rpm" {
        addArm("rpm", `SELECT 'rpm', rowid, name, version AS location, '' AS severity, rank FROM rpm_fts WHERE rpm_fts MATCH ? LIMIT ?`)
    }
    if p.Kind == "" || p.Kind == "deb" {
        addArm("deb", `SELECT 'deb', rowid, package_name, version AS location, '' AS severity, rank FROM deb_fts WHERE deb_fts MATCH ? LIMIT ?`)
    }
    if p.Kind == "" || p.Kind == "pypi" {
        addArm("pypi", `SELECT 'pypi', rowid, name, version AS location, '' AS severity, rank FROM pypi_fts WHERE pypi_fts MATCH ? LIMIT ?`)
    }
    if p.Kind == "" || p.Kind == "helm" {
        addArm("helm", `SELECT 'helm', rowid, chart_name, version AS location, '' AS severity, rank FROM helm_fts WHERE helm_fts MATCH ? LIMIT ?`)
    }

    if len(arms) == 0 {
        return nil, nil
    }

    fullQuery := strings.Join(arms, " UNION ALL ") + " ORDER BY score LIMIT ?"
    args = append(args, p.Limit)

    rows, err := db.Reader.QueryContext(ctx, fullQuery, args...)
    // ... scan rows into []SearchResult, return
}

func sanitizeFTS5Query(q string) string {
    // Escape special FTS5 characters, support prefix queries with *
    q = strings.TrimSpace(q)
    if q == "" {
        return ""
    }
    // For simple queries: wrap each word as a prefix match
    words := strings.Fields(q)
    for i, w := range words {
        // Strip quotes and special chars
        w = strings.ReplaceAll(w, `"`, ``)
        w = strings.ReplaceAll(w, `'`, ``)
        if w != "" {
            words[i] = `"` + w + `"` + "*"  // prefix match
        }
    }
    return strings.Join(words, " ")
}
```
