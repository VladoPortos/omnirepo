**1. Initialize the frontend project:**

```bash
cd web
npm create vite@latest . -- --template react-ts
```

If web/ already has content, initialize manually by creating package.json.

**2. Install all dependencies per D-34:**

```bash
cd web
npm install react@19.2.5 react-dom@19.2.5 react-router-dom@7.14.1
npm install @tanstack/react-query@5.99.0 framer-motion@12.38.0
npm install @milkdown/kit@7.20.0 @milkdown/react@7.20.0 @milkdown/theme-nord@7.20.0
npm install shiki@4.0.2 react-diff-viewer-continued@4.2.0

npm install -D @types/react @types/react-dom
npm install -D @playwright/test@1.59.1
```

**3. Configure Vite (`web/vite.config.ts`):**

```typescript
import { defineConfig } from 'vite';
import react from '@vitejs/plugin-react';
import tailwindcss from '@tailwindcss/vite';
import path from 'path';

export default defineConfig({
  plugins: [
    react(),
    tailwindcss(),
  ],
  resolve: {
    alias: {
      '@': path.resolve(__dirname, './src'),
    },
  },
  server: {
    port: 5173,
    proxy: {
      '/api': 'http://localhost:8080',
      '/v2': 'http://localhost:8080',
      '/s3': 'http://localhost:8080',
      '/git': 'http://localhost:8080',
    },
  },
  build: {
    outDir: 'dist',
    sourcemap: false,
  },
});
```

**1. Create Go embed file (`web/embed.go`):**

```go
package web

import "embed"

//go:embed dist/*
var DistFS embed.FS
```

This makes the built SPA available to the Go binary. The `internal/httpx/spa.go` handler (created in Plan 03) will use `web.DistFS` to serve the SPA.

**2. Copy Swagger UI assets into web/public/swagger/:**

Create a build script step that copies the needed swagger-ui-dist files:

```bash
mkdir -p web/public/swagger
cp web/node_modules/swagger-ui-dist/swagger-ui-bundle.js web/public/swagger/
cp web/node_modules/swagger-ui-dist/swagger-ui-standalone-preset.js web/public/swagger/
cp web/node_modules/swagger-ui-dist/swagger-ui.css web/public/swagger/
cp web/node_modules/swagger-ui-dist/favicon-16x16.png web/public/swagger/ 2>/dev/null || true
cp web/node_modules/swagger-ui-dist/favicon-32x32.png web/public/swagger/ 2>/dev/null || true
```

Create `web/public/swagger/index.html`:

```html
<!DOCTYPE html>
<html lang="en">
<head>
  <meta charset="UTF-8">
  <title>OmniRepo API Documentation</title>
  <link rel="stylesheet" href="swagger-ui.css" />
</head>
<body>
  <div id="swagger-ui"></div>
  <script src="swagger-ui-bundle.js"></script>
  <script src="swagger-ui-standalone-preset.js"></script>
  <script>
    window.onload = function() {
      SwaggerUIBundle({
        url: "/api/v1/openapi.yaml",
        dom_id: '#swagger-ui',
        deepLinking: true,
        presets: [
          SwaggerUIBundle.presets.apis,
          SwaggerUIStandalonePreset
        ],
        layout: "StandaloneLayout"
      });
    };
  </script>
</body>
</html>
```

**3. Add a `copy-swagger` npm script** in package.json under "scripts":

```json
"copy-swagger": "mkdir -p public/swagger && cp node_modules/swagger-ui-dist/swagger-ui-bundle.js node_modules/swagger-ui-dist/swagger-ui-standalone-preset.js node_modules/swagger-ui-dist/swagger-ui.css public/swagger/",
"prebuild": "npm run copy-swagger"
```

This ensures swagger assets are copied before every build.

**4. Rebuild to include swagger assets in dist/:**

```bash
cd web && npm run build
```

Verify `web/dist/swagger/index.html` exists in the built output.
