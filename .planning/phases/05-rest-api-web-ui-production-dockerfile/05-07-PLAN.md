**1. `web/src/pages/repo/RepoDetailRouter.tsx`:**

Reads URL params `:name`, `:type`, `:repo`. Fetches repo detail. Dispatches to the correct page component based on `repo.type`:

```tsx
export function RepoDetailRouter() {
  const { name, type, repo } = useParams();
  const { data, isLoading } = useRepo(name!, type!, repo!);
  if (isLoading) return <RepoSkeleton />;
  switch (data?.type) {
    case 'docker': return <DockerRepoPage repo={data} />;
    case 'rpm': return <RpmRepoPage repo={data} />;
    case 'deb': return <AptRepoPage repo={data} />;
    case 'pypi': return <PypiRepoPage repo={data} />;
    case 'helm': return <HelmRepoPage repo={data} />;
    case 'git': return <GitRepoPage repo={data} />;
    case 'raw': return <RawRepoPage repo={data} />;
    default: return <NotFoundPage />;
  }
}
```

Milkdown WYSIWYG editor wrapper:

```tsx
import { Editor, rootCtx, defaultValueCtx } from '@milkdown/kit/core';
import { commonmark } from '@milkdown/kit/preset/commonmark';
import { Milkdown, MilkdownProvider, useEditor } from '@milkdown/react';
import { nord } from '@milkdown/theme-nord';
import { listener, listenerCtx } from '@milkdown/kit/plugin/listener';

interface Props {
  value: string;
  onChange: (markdown: string) => void;
}

function MilkdownEditor({ value, onChange }: Props) {
  useEditor((root) =>
    Editor.make()
      .config(nord)
      .config((ctx) => {
        ctx.set(rootCtx, root);
        ctx.set(defaultValueCtx, value);
        ctx.get(listenerCtx).markdownUpdated((_, md) => onChange(md));
      })
      .use(commonmark)
      .use(listener),
  );
  return <Milkdown />;
}

export function MarkdownEditor(props: Props) {
  return (
    <MilkdownProvider>
      <MilkdownEditor {...props} />
    </MilkdownProvider>
  );
}
```

This is used in the Settings tab of every repo page for editing `description_md`.
