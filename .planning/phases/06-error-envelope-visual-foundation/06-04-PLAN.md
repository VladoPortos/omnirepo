var envelopeCodeRegex = regexp.MustCompile(`^[a-z][a-z0-9_]*\.[a-z][a-z0-9_]*$`)

func TestNormalizedCodesPassRegex(t *testing.T) {
    legacyCodes := []string{
        ErrPasswordChangeRequired,
        ErrUnauthenticated,
        ErrForbidden,
        ErrNotFound,
        ErrValidationFailed,
        ErrConflict,
        ErrInternal,
    }
    for _, code := range legacyCodes {
        t.Run(code, func(t *testing.T) {
            normalized := normalizeLegacyCode(code)
            if !envelopeCodeRegex.MatchString(normalized) {
                t.Errorf("normalizeLegacyCode(%q) = %q, does not match envelope code regex", code, normalized)
            }
        })
    }
}

func TestNormalizeLegacyCode_Table(t *testing.T) {
    cases := []struct {
        in, want string
    }{
        {"password-change-required", "auth.password_change_required"},
        {"unauthenticated", "auth.unauthenticated"},
        {"forbidden", "auth.forbidden"},
        {"not_found", "resource.not_found"},
        {"validation_failed", "validation.failed"},
        {"conflict", "resource.conflict"},
        {"internal", "api.internal"},
        {"some.dotted.code", "some.dotted.code"}, // pass-through
        {"UPPER-case", "legacy.upper_case"},      // sanitize + legacy prefix
        {"", "api.unknown"},                      // empty → sentinel
        {"123bad", "legacy.x_123bad"},            // leading-digit guard
    }
    for _, tc := range cases {
        t.Run(tc.in, func(t *testing.T) {
            got := normalizeLegacyCode(tc.in)
            if got != tc.want {
                t.Errorf("normalizeLegacyCode(%q) = %q, want %q", tc.in, got, tc.want)
            }
            if !envelopeCodeRegex.MatchString(got) {
                t.Errorf("result %q does not match regex", got)
            }
        })
    }
}

func TestInferClassFromStatus_Table(t *testing.T) {
    cases := []struct {
        status int
        want   string  // Class as string
    }{
        {400, "validation"},
        {404, "validation"},
        {409, "validation"},
        {413, "validation"},
        {422, "validation"},
        {401, "permission"},
        {403, "permission"},
        {429, "transient"},
        {500, "transient"},
        {502, "transient"},
        {503, "transient"},
    }
    for _, tc := range cases {
        t.Run("status_"+itoa(tc.status), func(t *testing.T) {
            got := string(inferClassFromStatus(tc.status))
            if got != tc.want {
                t.Errorf("inferClassFromStatus(%d) = %q, want %q", tc.status, got, tc.want)
            }
        })
    }
}

// itoa is a local helper to avoid importing strconv in test header.
func itoa(n int) string {
    if n == 0 {
        return "0"
    }
    s := ""
    for n > 0 {
        s = string(rune('0'+n%10)) + s
        n /= 10
    }
    return s
}

var (

    envelopeCodeRegex = regexp.MustCompile(`^[a-z][a-z0-9_]*\.[a-z][a-z0-9_]*$`)

type envelope struct {
    Code       string                 `json:"code"`
    Message    string                 `json:"message"`
    Hint       string                 `json:"hint,omitempty"`
    Class      string                 `json:"class"`
    IncidentID string                 `json:"incident_id,omitempty"`
    Details    map[string]interface{} `json:"details,omitempty"`
}

// assertEnvelope parses body as ApiErrorEnvelope and runs universal invariants.
func assertEnvelope(t *testing.T, body []byte) envelope {
    t.Helper()
    var e envelope
    if err := json.Unmarshal(body, &e); err != nil {
        t.Fatalf("body is not valid JSON: %v\nbody: %s", err, body)
    }
    if e.Code == "" || e.Message == "" || e.Class == "" {
        t.Fatalf("envelope missing required fields: %+v", e)
    }
    if !envelopeCodeRegex.MatchString(e.Code) {
        t.Errorf("code %q does not match regex", e.Code)
    }
    switch e.Class {
    case "validation", "permission", "transient", "operator_action_required":
        // ok
    default:
        t.Errorf("unknown class %q", e.Class)
    }
    // ERR-03: message and hint must not leak internals.
    if httperr.IsInternalString(e.Message) {
        t.Errorf("message leaks internal string: %q", e.Message)
    }
    if e.Hint != "" && httperr.IsInternalString(e.Hint) {
        t.Errorf("hint leaks internal string: %q", e.Hint)
    }
    // Extra regex checks.
    for _, bad := range []string{".go:", "goroutine ", "runtime.", "/usr/", "/home/", "/var/"} {
        if strings.Contains(e.Message, bad) || strings.Contains(e.Hint, bad) {
            t.Errorf("message or hint contains %q: %+v", bad, e)
        }
    }
    return e
}
