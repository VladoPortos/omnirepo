```tsx
/**
 * Dev-only story page rendering a matrix of StatusBadge variants
 * (6 statuses × 2 sizes × 2 iconOnly values) for Playwright snapshot
 * testing. Registered only when import.meta.env.DEV.
 * Phase 6 / plan 07.
 */

import { StatusBadge, type StatusVariant } from '@/components/common/StatusBadge';

const STATUSES: StatusVariant[] = [
  'healthy',
  'warning',
  'failure',
  'disabled',
  'maintenance',
  'neutral',
];
const SIZES = ['sm', 'md'] as const;
const ICON_ONLY = [false, true] as const;

export function StatusBadgeStoryPage() {
  return (
    <div className="p-8 space-y-6" data-story-root="status-badge">
      <h1 className="text-2xl font-semibold tracking-tight">
        StatusBadge matrix
      </h1>
      <p className="text-xs text-muted-foreground">
        6 statuses × 2 sizes × 2 iconOnly values = 24 variants. Playwright snapshots this page as a single baseline.
      </p>
      <div className="grid gap-6">
        {ICON_ONLY.map((iconOnly) => (
          <section key={`io-${iconOnly}`}>
            <h2 className="text-lg font-semibold mb-3">
              {iconOnly ? 'iconOnly = true' : 'iconOnly = false'}
            </h2>
            <div className="grid gap-4" style={{ gridTemplateColumns: `repeat(${STATUSES.length * SIZES.length}, auto)` }}>
              {STATUSES.flatMap((status) =>
                SIZES.map((size) => (
                  <div
                    key={`${status}-${size}-${iconOnly}`}
                    data-story-variant={status}
                    data-story-size={size}
                    data-story-icon-only={iconOnly ? 'true' : 'false'}
                    className="flex items-center justify-center p-2"
                  >
                    <StatusBadge
                      status={status}
                      label={status}
                      size={size}
                      iconOnly={iconOnly}
                    />
                  </div>
                )),
              )}
            </div>
          </section>
        ))}
      </div>
    </div>
  );
}

export default StatusBadgeStoryPage;
```
